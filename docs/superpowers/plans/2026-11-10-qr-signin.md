# QR-code Sign-in Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let accounts without a Xiaomi password (Google/Facebook/Apple) sign in by scanning a QR code with the Xiaomi Home app.

**Architecture:** `XiaomiCloud` gains `start_qr_login()` / `wait_qr_login()` (Xiaomi's long-poll login, as the token extractor does it); `FakeCloud` mirrors them. `LoginFlow` gets a QR branch that ends in the existing device discovery; `LoginDialog` gets a QR page. Sign-ins without a password store only the session, and `CloudMap` accepts that.

**Tech Stack:** Python 3.14, PyGObject (GTK 4.24, libadwaita 1.10), requests, micloud 0.6, pytest, the nested GNOME Shell check (`tools/nested-check.sh` + probe extension).

**Spec:** `docs/superpowers/specs/2026-11-10-qr-signin-design.md`

## Global Constraints

- Login tokens (`passToken`, `ssecurity`, `serviceToken`, the long-poll and `location` URLs) are never logged and never appear in exception texts.
- Every request on a `XiaomiCloud` session has a timeout (`_TimeoutSession`); the long poll uses `QR_POLL_TIMEOUT = (10, 10)` so a cancel is noticed within ~10 s.
- QR credentials: `username=""`, `password=""`, session in `cloud_session`; the keyring item format is unchanged.
- Copy: button "Sign in with the Xiaomi Home app"; caption "For accounts that use Google, Facebook or Apple"; page title "Scan with Xiaomi Home"; link "Open in browser instead"; status "Waiting for approval…"; expiry "The code expired." + button "Show a new code".
- Line length 100; ruff E F I UP B; no heredocs in shell steps.

import json

import pytest
import requests
from requests.cookies import RequestsCookieJar

from chippy.core.cloud import (
    CloudError,
    CloudUnreachable,
    QrLogin,
    QrLoginCancelled,
    QrLoginExpired,
    XiaomiCloud,
)

HOST = "https://account.xiaomi.com"
QR_IMG = "https://account.xiaomi.com/qr.png?secret=QRSECRET"
POLL = "https://lp.account.xiaomi.com/lp/poll?k=POLLSECRET"
LOGIN_URL = "https://account.xiaomi.com/login?qr=1"
STS = "https://sts.api.io.mi.com/sts?d=STSSECRET"


class Resp:
    def __init__(self, body="", status=200, content=b"", cookies=None):
        self.text = body if isinstance(body, str) else "&&&START&&&" + json.dumps(body)
        self.status_code = status
        self.content = content
        self.cookies = cookies or RequestsCookieJar()


class FakeXiaomi:
    """`polls` scripts the long poll: each item is a status code, an exception, or 'ok'."""

    def __init__(self, polls=("ok",), start=None, sts_token="STOKEN"):
        self.polls = list(polls)
        self.start = start or {"qr": QR_IMG, "loginUrl": LOGIN_URL, "lp": POLL, "timeout": 300}
        self.sts_token = sts_token
        self.cookies = RequestsCookieJar()
        self.headers = {}
        self.calls = []

    def get(self, url, params=None, timeout=None, **_kw):
        self.calls.append((url.split("?")[0], params, timeout))
        if url == f"{HOST}/longPolling/loginUrl":
            return Resp(self.start)
        if url == QR_IMG:
            return Resp(content=b"\x89PNGqr")
        if url == POLL:
            item = self.polls.pop(0)
            if isinstance(item, BaseException):
                raise item
            if item == "ok":
                return Resp({
                    "userId": 4242, "ssecurity": "SSEC", "cUserId": "CUID",
                    "passToken": "PASSSECRET", "location": STS,
                })
            return Resp("", status=item)
        if url == STS:
            jar = RequestsCookieJar()
            if self.sts_token:
                jar.set("serviceToken", self.sts_token)
            return Resp(cookies=jar)
        raise AssertionError(url)

    def close(self):
        pass


def _cloud(fake):
    cloud = XiaomiCloud("", "")
    cloud._init_session = lambda reset=False: setattr(cloud, "session", fake)
    return cloud


class Clock:
    def __init__(self):
        self.now = 0.0

    def __call__(self):
        return self.now

    def sleep(self, seconds):
        self.now += seconds


def test_start_returns_image_link_and_poll_url():
    fake = FakeXiaomi()
    qr = _cloud(fake).start_qr_login()
    assert qr == QrLogin(b"\x89PNGqr", LOGIN_URL, POLL, 300)

def test_wait_retries_timeouts_then_signs_in():
    fake = FakeXiaomi(polls=[requests.Timeout("t " + POLL), 408, "ok"])
    cloud, clock = _cloud(fake), Clock()
    qr = cloud.start_qr_login()
    cloud.wait_qr_login(qr, lambda: False, clock=clock, sleep=clock.sleep)
    state = cloud.session_state()
    assert state == {"user_id": "4242", "service_token": "STOKEN", "ssecurity": "SSEC"}
    poll_timeouts = [t for url, _p, t in fake.calls if url == POLL.split("?")[0]]
    assert poll_timeouts and all(t == (10, 10) for t in poll_timeouts)


def test_wait_expires_after_the_timeout():

        "qr": QR_IMG, "loginUrl": LOGIN_URL, "lp": POLL, "timeout": 30,

def test_wait_stops_when_cancelled():

```python
    # QR sign-in, as the maintained token extractor does it (PiotrMachowski's
    # Xiaomi-cloud-tokens-extractor, MIT): accounts without a Xiaomi password (Google,
    # Facebook, Apple) approve the login in the Xiaomi Home app.
    @_network_safe
    def start_qr_login(self) -> QrLogin:
        self._init_session(reset=True)
        response = self.session.get(
            f"{ACCOUNT_HOST}/longPolling/loginUrl",
            params={
                "_qrsize": "480",
                "qs": "%3Fsid%3Dxiaomiio%26_json%3Dtrue",
                "callback": "https://sts.api.io.mi.com/sts",
                "_hasLogo": "false",
                "sid": "xiaomiio",
                "serviceParam": "",
                "_locale": "en_GB",
                "_dc": str(int(time.time() * 1000)),
            },
        )
        data = _json_body(response.text) if response.status_code == 200 else {}
        qr, login_url, poll_url = data.get("qr"), data.get("loginUrl"), data.get("lp")
        if not (qr and login_url and poll_url):
            raise CloudError(f"Xiaomi didn't start a QR sign-in{_reason(data)}")
        image = self.session.get(qr)
        if image.status_code != 200 or not image.content:
            raise CloudError(f"couldn't download the QR code (HTTP {image.status_code})")
        try:
            timeout = int(data.get("timeout") or QR_DEFAULT_TIMEOUT)
        except (TypeError, ValueError):
            timeout = QR_DEFAULT_TIMEOUT
        return QrLogin(image.content, str(login_url), str(poll_url), timeout)

    def wait_qr_login(
        self,
        qr: QrLogin,
        cancelled: Callable[[], bool],
        clock: Callable[[], float] = time.monotonic,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        """Block until the QR login is approved (the session is then set), it expires
        (QrLoginExpired) or `cancelled()` turns true (QrLoginCancelled)."""
        deadline = clock() + qr.timeout
        while True:
            if cancelled():
                raise QrLoginCancelled()
            if clock() >= deadline:
                raise QrLoginExpired()
            try:
                response = self.session.get(qr.poll_url, timeout=QR_POLL_TIMEOUT)
            except requests.Timeout:
                sleep(0)  # let a test clock move; the request itself waited
                continue
            except requests.RequestException:
                raise CloudUnreachable() from None
            if response.status_code == 200:
                break
            sleep(1)  # nothing yet: don't spin
        if cancelled():
            raise QrLoginCancelled()
        data = _json_body(response.text)
        user_id, ssecurity, location = data.get("userId"), data.get("ssecurity"), data.get("location")
        if not (user_id and ssecurity and location):
            raise CloudError("Xiaomi's QR sign-in reply was incomplete")
        try:
            sts = self.session.get(location)
        except requests.RequestException:
            raise CloudUnreachable() from None
        token = sts.cookies.get("serviceToken") or self._cookie("serviceToken")
        if not token:
            raise CloudError("Xiaomi's QR sign-in finished without a service token")
        self.user_id = str(user_id)
        self.ssecurity = ssecurity
        self.cuser_id = data.get("cUserId")
        self.pass_token = data.get("passToken")
        self.service_token = token
```

```python
    def sign_in_qr(self) -> None:
        """QR sign-in: page `qr` with the code, then device discovery once approved."""
        if not self._start():
            return
        self.cancel_qr()  # a previous code's wait, if any
        self._username = self._password = ""
        self._cloud = self._factory("", "")
        cancel = {"set": False}
        self._qr_cancel = cancel
        cloud = self._cloud

        def started(qr, error):
            if cancel["set"]:
                return
            if error is not None:
                self._error("qr", _message(error))
                return
            self._on_page("qr", {"image": qr.image, "url": qr.login_url})
            self.busy = True  # waiting for approval: other submissions stay ignored
            self._runner.run(
                lambda: cloud.wait_qr_login(qr, lambda: cancel["set"]), approved
            )

        def approved(_r, error):
            if cancel["set"] or isinstance(error, QrLoginCancelled):
                self.busy = False
                return
            if isinstance(error, QrLoginExpired):
                self._on_page("error", {"page": "qr", "message": str(error), "expired": True})
            elif error is not None:
                self._error("qr", _message(error))
            else:
                self._discover()

        self._runner.run(cloud.start_qr_login, started)

    def cancel_qr(self) -> None:
        """Stop waiting for a QR approval (Back, close, or a new code)."""
        if self._qr_cancel is not None:
            self._qr_cancel["set"] = True
            self._qr_cancel = None
        self.busy = False
```
